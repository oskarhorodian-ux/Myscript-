local function getCustomMoveDirection()
    local cam = workspace.CurrentCamera
    local fwd = cam.CFrame.LookVector
    local right = cam.CFrame.RightVector
    fwd = Vector3.new(fwd.X, 0, fwd.Z)
    right = Vector3.new(right.X, 0, right.Z)
    if fwd.Magnitude > 0 then fwd = fwd.Unit end
    if right.Magnitude > 0 then right = right.Unit end
    local dir = Vector3.new(0, 0, 0)
    if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + fwd end
    if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - fwd end
    if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + right end
    if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - right end
    if dir.Magnitude > 0 then dir = dir.Unit end
    return dir
end

local function buildGhostClone(char, hrp)
    local ok, clone = pcall(function() return char:Clone() end)
    if not ok or not clone then return nil end

    clone.Name = "GhostVisual"

    for _, d in ipairs(clone:GetDescendants()) do
        if d:IsA("BasePart") then
            d.CanCollide = false
            d.Anchored = true
            d.CanQuery = false
            d.CanTouch = false
            d.CastShadow = false
            d.Material = Enum.Material.SmoothPlastic
            d.Transparency = 0.45
            if d.Name == "Handle" then d.Transparency = 0.45 end
        elseif d:IsA("Decal") or d:IsA("Texture") then
            d.Transparency = 0.45
        elseif d:IsA("BillboardGui") then
            d.Enabled = false
        elseif d:IsA("SurfaceGui") then
            d.Enabled = false
        elseif d:IsA("Accessory") then
            local handle = d:FindFirstChild("Handle")
            if handle and handle:IsA("BasePart") then
                handle.Transparency = 0.45
                handle.CanCollide = false
                handle.CastShadow = false
            end
        end
    end

    local cloneHum = clone:FindFirstChildOfClass("Humanoid")
    if cloneHum then
        cloneHum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
        cloneHum.HealthDisplayType = Enum.HumanoidHealthDisplayType.AlwaysOff
        cloneHum.NameDisplayDistance = 0
    end

    for _, d in ipairs(clone:GetDescendants()) do
        if d:IsA("BillboardGui") or d:IsA("TextLabel") then
            pcall(function() d:Destroy() end)
        end
    end

    ghostVisualClone = clone
    clone.Parent = workspace
    pcall(function() ghostVisualClone:SetPrimaryPartCFrame(hrp.CFrame) end)
    return clone
end

local function startPhantomSystem(mode)
    if phantomRunning then stopPhantomSystem(true) end

    local char = Player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not char or not hrp or not hum then return end

    char.Archivable = true
    phantomModeType = mode
    phantomRunning = true

    if mode == "invis" then
        invisActive = true
        godmodeEnabled = false
    else
        godmodeEnabled = true
        invisActive = false
    end

    destroyPhantomObjects()

    if mode == "invis" then
        local clone = buildGhostClone(char, hrp)
        if clone then
            local cam2 = workspace.CurrentCamera
            local target = hrp.Position + cam2.CFrame.LookVector * 5
            for _ = 1, 30 do
                local moving = getCustomMoveDirection()
                if moving.Magnitude > 0 then target = target + moving * 2 end
                task.wait(0.03)
            end
            if phantomRunning and ghostVisualClone and ghostVisualClone.PrimaryPart then
                ghostVisualClone:SetPrimaryPartCFrame(CFrame.new(target, target + cam2.CFrame.LookVector))
            end
        end
    end

    local hipOffset = (hum.HipHeight > 0 and hum.HipHeight or 2) + (hrp.Size.Y / 2)
    local spawnPos = hrp.Position - Vector3.new(0, hipOffset + 0.15, 0)

    if mode == "invis" and ghostVisualClone and ghostVisualClone.PrimaryPart then
        spawnPos = ghostVisualClone.PrimaryPart.Position
    end

    local cube = Instance.new("Part")
    cube.Name = "GhostSimulationCube"
    cube.Size = Vector3.new(2, 2, 1)
    cube.Transparency = 1
    cube.CanCollide = false
    cube.CanQuery = false
    cube.CanTouch = false
    cube.CastShadow = false
    cube.Anchored = true
    cube.CFrame = CFrame.new(spawnPos)
    cube.Parent = workspace
    ghostSimulationCube = cube
    godGroundCube = cube

    if mode == "invis" and ghostVisualClone and ghostVisualClone.PrimaryPart then
        ghostVisualClone.PrimaryPart.CFrame = cube.CFrame
    end

    hrp.CFrame = CFrame.new(hrp.Position + phantomSafeOffset)
    hrp.Anchored = true
    hum.WalkSpeed = 0

    local cam = workspace.CurrentCamera
    cam.CameraSubject = cube
    cam.CameraType = Enum.CameraType.Custom

    local camTarget = (mode == "invis" and ghostVisualClone) or cube
    local conns = {}
    local localVVel = 0
    local lastSafeY = spawnPos.Y

    local function addConn(c)
        table.insert(conns, c)
    end

    addConn(RunService.RenderStepped:Connect(function(dt)
        if not phantomRunning or _G.ScriptDestroyed then return end

        local cam2 = workspace.CurrentCamera
        if not spectatingPlayer and not freecamEnabled and camTarget and camTarget.Parent then
            cam2.CameraSubject = camTarget
        end

        if _G.ScriptPaused or isCastingSkill then return end

        local moving = getCustomMoveDirection()
        local pos = cube.Position
        local speed = walkSpeedValue * dt
        local newPos = pos + moving * speed

        local rp = RaycastParams.new()
        rp.FilterType = Enum.RaycastFilterType.Exclude
        rp.FilterDescendantsInstances = {char, cube, ghostVisualClone}
        local hit = workspace:Raycast(newPos + Vector3.new(0, 4, 0), Vector3.new(0, -80, 0), rp)

        local targetY
        if hit then
            targetY = hit.Position.Y + (hipOffset + 0.15)
            lastSafeY = targetY
            localVVel = 0
        else
            localVVel = localVVel + (workspace.Gravity * dt * 0.55)
            targetY = pos.Y - localVVel * dt
        end

        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
            localVVel = 0
            targetY = pos.Y + (jumpPowerValue * dt * 2.2)
        end

        local finalPos = Vector3.new(newPos.X, targetY, newPos.Z)
        cube.CFrame = CFrame.new(finalPos, finalPos + cam2.CFrame.LookVector)

        if mode == "invis" and ghostVisualClone and ghostVisualClone.PrimaryPart then

        end

            hrp.CFrame = CFrame.new(hrp.Position + phantomSafeOffset)
        end
    end))

    addConn(RunService.Heartbeat:Connect(function()
        if not phantomRunning or _G.ScriptDestroyed then return end
        if _G.ScriptPaused then return end
        if hum.Health <= 0 then
            stopPhantomSystem(false)
        end
    end))

    task.spawn(function()
        while phantomRunning and not _G.ScriptDestroyed do task.wait(0.5) end
        for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
    end)

local function startFlyPhysics()
    if flyLoopRunning then return end
    local char = Player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return end

    hum.PlatformStand = true
    flyConn = RunService.RenderStepped:Connect(function()
        if not flying or _G.ScriptDestroyed then return end
        if _G.ScriptPaused then return end
        local camera = workspace.CurrentCamera
        local dir = Vector3.new(0, 0, 0)
        if keysPressed[Enum.KeyCode.W] then dir = dir + camera.CFrame.LookVector end
        if keysPressed[Enum.KeyCode.S] then dir = dir - camera.CFrame.LookVector end
        if keysPressed[Enum.KeyCode.D] then dir = dir + camera.CFrame.RightVector end
        if keysPressed[Enum.KeyCode.A] then dir = dir - camera.CFrame.RightVector end
        if keysPressed[Enum.KeyCode.Space] then dir = dir + Vector3.new(0, 1, 0) end
        if keysPressed[Enum.KeyCode.LeftShift] then dir = dir - Vector3.new(0, 1, 0) end
        if dir.Magnitude > 0 then
            hrp.CFrame = hrp.CFrame + dir.Unit * flySpeed * (1 / 60)
        end
    end)
end

local function getScreenCenterTarget(maxDist)
    local cam = workspace.CurrentCamera
    local vp = cam.ViewportSize
    local center = Vector2.new(vp.X / 2, vp.Y / 2)
    local char = Player.Character
    local myHrp = char and char:FindFirstChild("HumanoidRootPart")
    if not myHrp then return nil end

    local best, bestScore = nil, math.huge
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= Player and plr.Character then
            local eHum = plr.Character:FindFirstChildOfClass("Humanoid")
            local eHrp = plr.Character:FindFirstChild("HumanoidRootPart")
            if eHum and eHrp and eHum.Health > 0 then
                local dist = (eHrp.Position - myHrp.Position).Magnitude
                if dist <= maxDist then
                    local sp, onScreen = cam:WorldToViewportPoint(eHrp.Position)
                    if onScreen then
                        local score = (Vector2.new(sp.X, sp.Y) - center).Magnitude + dist * 0.03
                        if score < bestScore then
                            bestScore = score
                            best = plr
                        end
                    end
                end
            end
        end
    end
    return best
end

local function aimStep()
    if not aimEnabled or _G.ScriptPaused or _G.ScriptDestroyed then return end
    local char = Player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hrp then return end

    aimCurrentTarget = getScreenCenterTarget(aimMaxDist)
    if not aimCurrentTarget or not aimCurrentTarget.Character then return end

    local tHrp = aimCurrentTarget.Character:FindFirstChild("HumanoidRootPart")
    if not tHrp then return end

    local cam = workspace.CurrentCamera
    local camPos = cam.CFrame.Position
    local dir = tHrp.Position - camPos
    if dir.Magnitude < 0.5 then return end

    local goal = CFrame.new(camPos, camPos + dir.Unit)
    cam.CFrame = cam.CFrame:Lerp(goal, math.clamp(aimSmoothValue, 0.05, 1))

    if hum then
        hum.AutoRotate = false
        hrp.CFrame = CFrame.new(hrp.Position, Vector3.new(tHrp.Position.X, hrp.Position.Y, tHrp.Position.Z))
    end
end

                    local box = Instance.new("BoxHandleAdornment")
                    box.Name = "PH_Hitbox"

                    box.Size = hrp.Size + Vector3.new(0.5, 0.5, 0.5)

                    box.ZIndex = 5
                    box.Transparency = 0.55
                    box.Color3 = Color3.fromRGB(255, 60, 60)
                    box.Parent = hrp
                    hitboxAdornments[hrp] = box

                    local label = Instance.new("BillboardGui")

local function evaluateMathExpression(expr)
    if not expr then return nil end
    local cleaned = tostring(expr)
    cleaned = string.gsub(cleaned, "х", "*")
    cleaned = string.gsub(cleaned, "Х", "*")
    cleaned = string.gsub(cleaned, "÷", "/")
    cleaned = string.gsub(cleaned, ":", "/")
    cleaned = string.gsub(cleaned, "×", "*")
    cleaned = string.gsub(cleaned, ",", ".")
    cleaned = string.gsub(cleaned, "[^%d%+%-%*/%.%(%)%s]", "")
    if cleaned == "" then return nil end
    if not string.find(cleaned, "[%+%-%*/]") then return nil end
    local loader = loadstring or load
    if not loader then return nil end
    local ok, fn = pcall(loader, "return " .. cleaned)
    if not ok or not fn then return nil end
    local ok2, val = pcall(fn)
    if not ok2 or type(val) ~= "number" then return nil end
    if val ~= val then return nil end
    if math.abs(val) > 1e15 then return nil end
    return val
end

local function buildBoxVisual(bData)
    local a = bData.PointA
    local b = bData.PointB
    local mid = (a + b) / 2
    local sizeX = math.abs(a.X - b.X) + 10
    local sizeY = math.abs(a.Y - b.Y) + 20
    local sizeZ = math.abs(a.Z - b.Z) + 10

    local part = Instance.new("Part")
    part.Name = "ZoneBox"
    part.Size = Vector3.new(sizeX, sizeY, sizeZ)
    part.Position = mid
    part.Anchored = true
    part.CanCollide = false
    part.CanQuery = false
    part.Transparency = 0.85
    part.Color = getBoxColorByRule(bData.Rule)
    part.Material = Enum.Material.ForceField
    part.Parent = workspace

    local bbg = Instance.new("BillboardGui")
    bbg.Name = "BoxTitleGui"
    bbg.Size = UDim2.new(0, 200, 0, 40)
    bbg.StudsOffset = Vector3.new(0, sizeY / 2 + 2, 0)
    bbg.AlwaysOnTop = true
    bbg.Adornee = part
    bbg.Parent = part

    local lbl = Instance.new("TextLabel")
    lbl.Name = "TitleLbl"
    lbl.Size = UDim2.new(1, 0, 1, 0)
    lbl.BackgroundTransparency = 1
    lbl.Text = bData.Name .. " [" .. bData.Rule .. "]"
    lbl.TextColor3 = getBoxColorByRule(bData.Rule)
    lbl.TextStrokeTransparency = 0
    lbl.TextSize = 14
    lbl.Font = Enum.Font.GothamBold
    lbl.Parent = bbg

    bData.Visual = part

    task.spawn(function()
        while part and part.Parent and not _G.ScriptDestroyed do
            local cam = workspace.CurrentCamera
            local dist = (cam.CFrame.Position - part.Position).Magnitude
            lbl.TextSize = math.clamp(math.floor(300 / math.max(dist, 1)), 8, 18)
            task.wait(0.3)
        end
    end)

    return part
end

    end
end

local function finalizeNewBox()
    if not draftPointA or not draftPointB then
        statusLbl.Text = "Нужны обе точки"
        return
    end
    if nameBox.Text == "" and not editingBoxData then
        statusLbl.Text = "Введи название"
        return
    end

    if editingBoxData then
        editingBoxData.PointA = draftPointA
        editingBoxData.PointB = draftPointB
        editingBoxData.Name = nameBox.Text
        editingBoxData.Rule = ruleValue
        if editingBoxData.Visual then editingBoxData.Visual:Destroy() end
        buildBoxVisual(editingBoxData)
        editingBoxData = nil
    else
        local newData = {
            Name = nameBox.Text,
            PointA = draftPointA,
            PointB = draftPointB,
            Rule = ruleValue,
            Visual = nil
        }
        table.insert(_G.SavedZoneBoxes, newData)
        buildBoxVisual(newData)
    end

    if draftMarkerA then draftMarkerA:Destroy() draftMarkerA = nil end
    if draftMarkerB then draftMarkerB:Destroy() draftMarkerB = nil end
    draftPointA = nil
    draftPointB = nil
    nameBox.Text = ""
    ruleValue = "+"
    updateCreatorUI()
    refreshBoxesList()
end

local function isInsideBox(pos, bData)
    local minX = math.min(bData.PointA.X, bData.PointB.X)
    local maxX = math.max(bData.PointA.X, bData.PointB.X)
    local minY = math.min(bData.PointA.Y, bData.PointB.Y) - 10
    local maxY = math.max(bData.PointA.Y, bData.PointB.Y) + 10
    local minZ = math.min(bData.PointA.Z, bData.PointB.Z)
    local maxZ = math.max(bData.PointA.Z, bData.PointB.Z)
    return pos.X >= minX and pos.X <= maxX and pos.Y >= minY and pos.Y <= maxY and pos.Z >= minZ and pos.Z <= maxZ
end

local function clampToBox(pos, bData)
    local minX = math.min(bData.PointA.X, bData.PointB.X) + 1.5
    local maxX = math.max(bData.PointA.X, bData.PointB.X) - 1.5
    local minY = math.min(bData.PointA.Y, bData.PointB.Y) - 10
    local maxY = math.max(bData.PointA.Y, bData.PointB.Y) + 10
    local minZ = math.min(bData.PointA.Z, bData.PointB.Z) + 1.5
    local maxZ = math.max(bData.PointA.Z, bData.PointB.Z) - 1.5
    return Vector3.new(
        math.clamp(pos.X, minX, maxX),
        math.clamp(pos.Y, minY, maxY),
        math.clamp(pos.Z, minZ, maxZ)
    )
end

    if not enterPressed then return end
    local n = tonumber(zipSpeedBox.Text)
    if n then
        zipSpeed = math.clamp(n, 1, 500)
        zipSpeedBox.Text = "Скор: " .. math.floor(zipSpeed)
    end
end)

local function recalculateOrders()
    local counter = 1
    for _, pt in ipairs(SavedPoints) do
        if pt.Order and pt.Order > 0 then
            pt.Order = counter
            counter = counter + 1
        end
    end
end

local function moveToDest(hrp, dest, abortFn)
    if not hrp or not hrp.Parent then return end
    local wasAnc = hrp.Anchored
    local useTeleport = true

    if useTeleport then
        hrp.CFrame = CFrame.new(dest)
        if not invisActive then
            task.wait(0.1)
        end
        return
    end

    while hrp and hrp.Parent and not _G.ScriptDestroyed do
        if abortFn and abortFn() then break end
        local diff = dest - hrp.Position
        if diff.Magnitude < 2 then break end
        hrp.CFrame = CFrame.new(hrp.Position + diff.Unit * math.min(zipSpeed / 60, diff.Magnitude))
        RunService.Heartbeat:Wait()
    end
    if hrp and hrp.Parent and not invisActive then hrp.Anchored = wasAnc end
end

local function bindDeathTracker(char)
    local hum = char:WaitForChild("Humanoid", 5)
    if not hum then return end
    hum.Died:Connect(function()
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if hrp then lastDeathPos = hrp.Position end
    end)
end

local function runZipline()
    if ziplineActive then
        ziplineActive = false
        return
    end
    local ordered = {}
    for _, pt in ipairs(SavedPoints) do
        if pt.Order and pt.Order > 0 then table.insert(ordered, pt) end
    end
    if #ordered == 0 then
        massActionsBtn.Text = "Нет точек с номерами"
        task.delay(2, function()
            if massActionsBtn.Text == "Нет точек с номерами" then
                massActionsBtn.Text = "⚙ Массовые действия"
            end
        end)
        return
    end
    table.sort(ordered, function(a, b) return a.Order < b.Order end)
    ziplineActive = true
    task.spawn(function()
        repeat
            for _, pt in ipairs(ordered) do
                if not ziplineActive or _G.ScriptDestroyed then break end
                local char = Player.Character
                local hrp = char and char:FindFirstChild("HumanoidRootPart")
                if hrp then
                    if invisActive and ghostSimulationCube then
                        ghostSimulationCube.CFrame = CFrame.new(pt.Pos + Vector3.new(0, 3, 0))
                        task.wait(0.12)
                    elseif godmodeEnabled and godGroundCube then
                        godGroundCube.CFrame = CFrame.new(pt.Pos + Vector3.new(0, 3, 0))
                        task.wait(0.12)
                    else
                        moveToDest(hrp, pt.Pos + Vector3.new(0, 3, 0), function() return not ziplineActive end)
                    end
                end
            end
        until not loopPointsEnabled or not ziplineActive or _G.ScriptDestroyed
        ziplineActive = false
    end)
end

        end
    end)
end)
